#Ansible - Install anf=d configure Nginx



- name: Install and config Nginx 
  hosts: all 
  become: yes

   task
      - name: Install nginx 
          yum: 
            name; nginx 
	    state:present

      - name: Configure niginx 
          template:
	   src: nginx.conf.j2 
	   dest: /etc/nginx/nginx.cong


      - name: Start nginx service 
          service:
	    name: nginx 
	    state: started
	    enable: yes
