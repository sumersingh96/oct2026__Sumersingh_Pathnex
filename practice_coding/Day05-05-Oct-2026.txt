#Ansible - Create User and set Permission


- name: create a new user
  host: all
  become: yes
  task:
     - name: Create user
       user:
         name: pathnex
	 state: present
	 shell: /bin/bash
     - name: Set permissions for the user
       filr:
         path: /home/pathnex
	 owner:pathnex
	 group:pathnex
	 mode: '0755'
