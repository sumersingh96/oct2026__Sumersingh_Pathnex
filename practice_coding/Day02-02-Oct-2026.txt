#Ansible - Install & Enable Nginx

- name: Install and restart Nginx on Pathnex 
  hosts: all 
  become: yes

  task:
  - name: Install nginx 
     yum:
      name: nginx 
      state: present
 
  - name: Enable nginx
     service:
      name: nnginx 
      state:started 
      enable: yes
